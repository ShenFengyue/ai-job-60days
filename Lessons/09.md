### `main.py`

```
import json
from pathlib import Path

import faiss
import numpy as np
from sentence_transformers import SentenceTransformer


MODEL_NAME = "sentence-transformers/all-MiniLM-L6-v2"
DATA_PATH = Path("data.json")
INDEX_PATH = Path("vectors.index")
RESULTS_PATH = Path("results.json")


def load_records():
    return json.loads(DATA_PATH.read_text(encoding="utf-8"))


def main():
    records = load_records()
    texts = [item["text"] for item in records]

    model = SentenceTransformer(MODEL_NAME)

    embeddings = model.encode(
        texts,
        convert_to_numpy=True,
        normalize_embeddings=True,
    ).astype("float32")

    dimension = embeddings.shape[1]

    # 归一化向量使用内积，可近似表示 cosine similarity
    index = faiss.IndexFlatIP(dimension)
    index.add(embeddings)

    faiss.write_index(index, str(INDEX_PATH))

    queries = [
        "How does exercise affect emotional well-being?",
        "What is a reusable block of Python code?",
        "How are vectors used for search?",
        "What improves performance?",
    ]

    all_results = []

    for query in queries:
        query_embedding = model.encode(
            [query],
            convert_to_numpy=True,
            normalize_embeddings=True,
        ).astype("float32")

        scores, indices = index.search(query_embedding, k=3)

        matches = []
        for score, idx in zip(scores[0], indices[0]):
            matches.append(
                {
                    "id": records[int(idx)]["id"],
                    "text": records[int(idx)]["text"],
                    "topic": records[int(idx)]["topic"],
                    "score": round(float(score), 4),
                }
            )

        all_results.append(
            {
                "query": query,
                "matches": matches,
            }
        )

    RESULTS_PATH.write_text(
        json.dumps(all_results, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )

    print(json.dumps(all_results, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
```

### 必须完成的 3 个实验

1. **索引保存与加载**
   程序先写出 `vectors.index`，再用：

   ```
   loaded_index = faiss.read_index("vectors.index")
   ```

   

   重新查询一次，确认结果与原索引一致。

2. **top-k 对比**
   分别测试 `k=1`、`k=3`、`k=5`，记录结果变化。你要理解：`k` 不是“准确率”，只是返回多少个候选。

3. **归一化对比**
   复制一份实验，把 `normalize_embeddings=True` 去掉，比较 `IndexFlatIP` 下的排序变化，并在 README 中说明为什么归一化会影响内积结果。
