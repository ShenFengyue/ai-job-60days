### 第一步：定义数据模型

```
models.py
from pydantic import BaseModel, Field


class WordCard(BaseModel):
    word: str = Field(min_length=1)
    meaning: str = Field(min_length=1)
    examples: list[str] = Field(min_length=1, max_length=3)
    mnemonic: str = Field(min_length=1)
    related_words: list[str] = Field(default_factory=list, max_length=5)
```

### 第二步：封装模型调用

```
llm_client.py
import json
import os

import requests
from dotenv import load_dotenv

from models import WordCard

load_dotenv()

API_KEY = os.getenv("DEEPSEEK_API_KEY")
MODEL = os.getenv("DEEPSEEK_MODEL", "deepseek-v4-flash")
URL = "https://api.deepseek.com/chat/completions"


def generate_word_card(word: str) -> WordCard:
    word = word.strip()

    if not word:
        raise ValueError("请输入一个英文单词")

    if not API_KEY:
        raise RuntimeError("缺少 DEEPSEEK_API_KEY")

    system_prompt = """
You are an English vocabulary teacher.

Return JSON only.
The JSON object must contain:
- word: string
- meaning: string
- examples: array of 1 to 3 strings
- mnemonic: string
- related_words: array of strings

Example:
{
  "word": "abandon",
  "meaning": "放弃；遗弃",
  "examples": ["He abandoned the plan."],
  "mnemonic": "把 a band on 想成把乐队放在一边。",
  "related_words": ["give up", "desert"]
}
"""

    payload = {
        "model": MODEL,
        "messages": [
            {"role": "system", "content": system_prompt},
            {
                "role": "user",
                "content": f"Please explain the English word: {word}",
            },
        ],
        "response_format": {"type": "json_object"},
        "temperature": 0.2,
        "max_tokens": 500,
        "stream": False,
    }

    response = requests.post(
        URL,
        headers={
            "Authorization": f"Bearer {API_KEY}",
            "Content-Type": "application/json",
        },
        json=payload,
        timeout=30,
    )
    response.raise_for_status()

    data = response.json()
    raw_content = data["choices"][0]["message"].get("content")

    if not raw_content:
        raise RuntimeError("模型返回了空内容")

    parsed = json.loads(raw_content)
    return WordCard.model_validate(parsed)
```

### 第三步：制作网页界面

```
app.py
import gradio as gr

from llm_client import generate_word_card


def explain_word(word: str):
    try:
        card = generate_word_card(word)

        examples = "\n".join(
            f"{index + 1}. {example}"
            for index, example in enumerate(card.examples)
        )

        related_words = ", ".join(card.related_words) or "暂无"

        return (
            f"单词：{card.word}\n\n"
            f"释义：{card.meaning}\n\n"
            f"例句：\n{examples}\n\n"
            f"记忆法：{card.mnemonic}\n\n"
            f"相关词：{related_words}"
        )

    except Exception as error:
        return f"发生错误：{error}"


demo = gr.Interface(
    fn=explain_word,
    inputs=gr.Textbox(
        label="输入英文单词",
        placeholder="例如：resilient",
    ),
    outputs=gr.Markdown(label="词汇卡片"),
    title="English Learning Assistant",
    description="输入一个英文单词，生成释义、例句、记忆法和相关词。",
)

if __name__ == "__main__":
    demo.launch()
```
